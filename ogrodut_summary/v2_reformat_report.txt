DIV_TOK = re.compile(r'<(/?)div\b[^>]*>', re.I)
TITLE = re.compile(r'<title>(.*?)</title>', re.S | re.I)
H1_MAIN = re.compile(r'<h1\s+class="main-title"\s*>(.*?)</h1>', re.S | re.I)
CARD_TAG = re.compile(r'<div class="card[^"]*" id="c([^"]*)"([^>]*)>')


def esc(s):
    return html.escape(s, quote=True)


def strip_tags(s):
    return re.sub(r'\s+', ' ', html.unescape(re.sub(r'<[^>]*>', ' ', s))).strip()


def div_inner(text, start_pat, from_pos=0):
    """start_pat er prothom match er <div> er balanced vitorer html + end index."""
    m = re.compile(start_pat, re.S | re.I).search(text, from_pos)
    if not m:
        return None, from_pos
    depth, pos = 1, m.end()
    for t in DIV_TOK.finditer(text, pos):
        depth += -1 if t.group(1) else 1
        if depth == 0:
            return text[pos:t.start()], t.end()
    return text[pos:], len(text)


def balance_divs(s):
    """Stray </div> bad dey, ghaton thakle shesh e bondho kore (page vange na)."""
    out, pos, depth = [], 0, 0
    for m in DIV_TOK.finditer(s):
        if m.group(1):
            if depth == 0:
                out.append(s[pos:m.start()])
                pos = m.end()
                continue
            depth -= 1
        else:
            depth += 1
    out.append(s[pos:])
    return "".join(out) + "</div>" * depth

def build_card(c, prev_id, next_id):
    cid = esc(c["id"])
    flag = c.get("flag", "") if c.get("flag", "") in COLORS else ""
    attrs = f' data-flag="{flag}"' if flag else ""
    if c.get("resume"):
        attrs += ' data-resume=""'
    if c.get("genre"):
        attrs += f' data-genre="{esc(c["genre"])}"'
    search = esc((c["id"] + " " + strip_tags(c["title_html"]) + " " + strip_tags(c["sub_html"])[:400]
                  + " " + c.get("genre", "")).lower())
    cls = "card resume" if c.get("resume") else "card"
    radios = "".join(
        f'<input type="radio" class="fr fr-{col}" name="f{cid}" id="f{cid}-{col}" value="{col}"'
        f'{" checked" if (flag or "none") == col else ""}>'
        for col in COLORS + ("none",))
    genre = f'<span class="genre">{esc(c["genre"])}</span>' if c.get("genre") else ""
    sub = f'<div class="{c["sub_class"]}"><!--s-->{c["sub_html"]}<!--/s--></div>' if c["sub_html"] else ""
    pn = ""
    if prev_id:
        pn += f'<a href="#c{esc(prev_id)}">← Prev</a>'
    if next_id:
        pn += f'<a href="#c{esc(next_id)}">Next →</a>'
    return f"""
<div class="{cls}" id="c{cid}" data-id="{cid}" data-search="{search}"{attrs}>
{radios}<span class="tint"></span>
<details class="cd" open>
<summary class="head"><span class="ind"></span><span class="qnum">{esc(c["label"])}</span>{genre}<h2><!--t-->{c["title_html"]}<!--/t--></h2>{sub}</summary>
<div class="body"><!--b-->{c["body"]}<!--/b--></div>
</details>
<div class="controls">
<label for="f{cid}-red" class="dot red"></label><label for="f{cid}-yellow" class="dot yellow"></label><label for="f{cid}-green" class="dot green"></label><label for="f{cid}-blue" class="dot blue"></label><label for="f{cid}-none" class="dot none"></label>
<button class="b-orange js-only" onclick="saveResume(this)">Resume Here</button>
<span class="nav"><span class="pn">{pn}</span><a href="#toc">☰ Index</a><a href="#top">↑ Top</a></span>
</div>
</div>"""

def _between(seg, k):
    m = re.search(rf'<!--{k}-->(.*?)<!--/{k}-->', seg, re.S)
    return m.group(1) if m else ""


def parse_v2(src):
    tags = list(CARD_TAG.finditer(src))
    cards = []
    for i, m in enumerate(tags):
        end = tags[i + 1].start() if i + 1 < len(tags) else len(src)
        seg = src[m.end():end]
        attrs = dict(re.findall(r'data-([a-z]+)="([^"]*)"', m.group(2)))
        lab = re.search(r'<span class="qnum">(.*?)</span>', seg, re.S)
        sc = re.search(r'<div class="(orig|by)"><!--s-->', seg)
        cards.append(dict(
            id=html.unescape(m.group(1)),
            label=html.unescape(lab.group(1)) if lab else "#" + m.group(1),
            title_html=_between(seg, "t"),
            sub_html=_between(seg, "s"),
            sub_class=sc.group(1) if sc else "by",
            genre=html.unescape(attrs.get("genre", "")),
            body=_between(seg, "b"),
            flag=attrs.get("flag", ""),
            resume="resume" in attrs))
    return cards

def apply_edits(cards, edits):
    if edits.get("resume") is not None:
        hit = False
        for c in cards:
            c["resume"] = (c["id"] == edits["resume"])
            hit = hit or c["resume"]
        if not hit:
            print(f"  ! resume id {edits['resume']} paoa jayni")
    for cid, col in edits.get("flags", {}).items():
        for c in cards:
            if c["id"] == cid:
                c["flag"] = "" if col == "none" else col
                break
        else:
            print(f"  ! flag id {cid} paoa jayni")


def convert(src, path, edits):
    r = load(src, path)
    if r is None:
        return None, 0
    cards, page_title, h1_html = r
    if not cards:
        return None, 0
    apply_edits(cards, edits)
    prefix = PREFIX_TAG + hashlib.md5(path.name.encode("utf-8")).hexdigest()[:8] + "_"
    light = bool(re.search(r'<input[^>]*\bid="lm"[^>]*\bchecked', src))   # baked theme thake
    return build_page(cards, page_title, h1_html, prefix, light), len(cards)


def process_file(path, dry, backup, edits):
    try:
        src = path.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return "SKIP (utf-8 noy)", 0
    page, n = convert(src, path, edits)
    if page is None:
        return "SKIP (format mile ni)", 0
    if page == src:
        return "UNCHANGED", n
    if dry:
        return "WOULD UPDATE", n
    if backup:
        shutil.copy2(path, path.with_suffix(path.suffix + ".bak"))
    tmp = path.with_name(path.name + ".tmp~")   # atomic in-place write, kono extra file thake na
    try:
        tmp.write_text(page, encoding="utf-8")
        tmp.replace(path)
    finally:
        if tmp.exists():
            tmp.unlink()
    return "UPDATED", n

GENERATOR = "reformat_index_docs_v2"
PREFIX_TAG = "idx_"
GOTO_PH = "Index"
DESCRIPTION = "'Index: N — Author >>> Title' type HTML - folder-wide in-place reformat"

SEP = re.compile(r'(?:-{5,})?\s*<h3[^>]*>\s*Index:\s*(\d+)\s*(?:&#8212;|&mdash;|—|–|-)\s*(.*?)</h3>', re.S | re.I)
BODY = re.compile(r'<body[^>]*>(.*)</body>', re.S | re.I)
BLOCKS = re.compile(r'<(style|script)\b.*?</\1>', re.S | re.I)
WRAP_TAGS = re.compile(r'</?(?:!DOCTYPE|html|head|body|meta|title|link)\b[^>]*>', re.I)
INLINE_STYLE = re.compile(r'\sstyle="[^"]*"', re.I)
LEGACY_TAG = re.compile(r'<div\s+class="card[^"]*"\s+data-id="([^"]*)"((?:\s+data-[a-z]+="[^"]*")*)\s*>', re.I)


def split_title(raw):
    """'Author >>> Title (Genre)' -> (author, title, genre). Nested bracket handle kore."""
    s = html.unescape(re.sub(r'<.*?>', '', raw)).strip()
    author, rest = (s.split(">>>", 1) if ">>>" in s else ("", s))
    author, rest = author.strip(), rest.strip()
    genre, title = "", rest
    if rest.endswith(")"):
        depth = 0
        for i in range(len(rest) - 1, -1, -1):
            if rest[i] == ")":
                depth += 1
            elif rest[i] == "(":
                depth -= 1
                if depth == 0:
                    genre, title = rest[i + 1:-1].strip(), rest[:i].strip()
                    break
    return author, title, genre


def clean_body(seg):
    b = BODY.search(seg)
    body = b.group(1) if b else seg            # <body> na thakle (fragment) puro segment
    body = BLOCKS.sub("", body)
    body = WRAP_TAGS.sub("", body)
    body = INLINE_STYLE.sub("", body)          # inline color/background theme vange
    body = body.replace("\\'", "'").strip()
    return balance_divs(body)


def make_card(idx, author, title, genre, body):
    return dict(id=idx, label=f"Index {idx}", title_html=esc(title),
                sub_html=(f"✍ {esc(author)}" if author else ""), sub_class="by",
                genre=genre, body=body, flag="", resume=False)


def parse_original(src):
    ms = list(SEP.finditer(src))
    out = []
    for i, m in enumerate(ms):
        end = ms[i + 1].start() if i + 1 < len(ms) else len(src)
        author, title, genre = split_title(m.group(2))
        out.append(make_card(m.group(1), author, title, genre, clean_body(src[m.end():end])))
    return out


def parse_legacy(src):
    """Ager version (card expanded + data-author/title/genre) theke migrate."""
    out = []
    for m in LEGACY_TAG.finditer(src):
        a = {k: html.unescape(v) for k, v in re.findall(r'data-([a-z]+)="([^"]*)"', m.group(0))}
        inner, _ = div_inner(src, re.escape(m.group(0)), m.start())
        body, _ = div_inner(inner or "", r'<div\s+class="details"\s*>')
        out.append(make_card(m.group(1), a.get("author", ""), a.get("title", ""), a.get("genre", ""),
                             (body or "").strip()))
    return out


def load(src, path):
    if f'name="generator" content="{GENERATOR}"' in src:
        cards = parse_v2(src)
        tm = TITLE.search(src)
        title = html.unescape(tm.group(1)).strip() if tm else path.stem
        mm = H1_MAIN.search(src)
        h1 = mm.group(1).strip() if mm else esc(title)
    elif 'name="generator" content="reformat_index_docs"' in src or ('data-search=' in src and 'data-genre=' in src and LEGACY_TAG.search(src)):
        cards = parse_legacy(src)
        tm = TITLE.search(src)
        title = html.unescape(tm.group(1)).strip() if tm else path.stem
        h1 = esc(title)
    else:
        cards = parse_original(src)
        title = path.stem.replace("_", " ")
        h1 = esc(title)
    if not cards:
        return None
    return cards, title, h1
