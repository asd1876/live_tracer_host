SEP = re.compile(
    r'(?:-{5,})?\s*<h3[^>]*>\s*Index:\s*(\d+)\s*(?:&#8212;|&mdash;|—|–|-)\s*(.*?)</h3>',
    re.S | re.I)
BODY = re.compile(r'<body[^>]*>(.*)</body>', re.S | re.I)
BLOCKS = re.compile(r'<(style|script)\b.*?</\1>', re.S | re.I)
WRAP_TAGS = re.compile(r'</?(?:!DOCTYPE|html|head|body|meta|title|link)\b[^>]*>', re.I)
INLINE_STYLE = re.compile(r'\sstyle="[^"]*"', re.I)
TITLE = re.compile(r'<title>(.*?)</title>', re.S | re.I)
DIV_TOK = re.compile(r'<(/?)div\b[^>]*>', re.I)


# ----------------------------------------------------------------- helpers
def balance_divs(s):
    """Stray </div> bad dei, ghaton thakle shesh e </div> diye bondho kore (page vange na)."""
    out, pos, depth = [], 0, 0
    for m in DIV_TOK.finditer(s):
        if m.group(1):  # closing
            if depth == 0:
                out.append(s[pos:m.start()])
                pos = m.end()
                continue
            depth -= 1
        else:
            depth += 1
    out.append(s[pos:])
    return "".join(out) + "</div>" * depth


def split_title(raw):
    """'Author >>> Title (Genre)' -> (author, title, genre). Nested bracket handle kore."""
    s = html.unescape(re.sub(r'<.*?>', '', raw)).strip()
    if ">>>" in s:
        author, rest = s.split(">>>", 1)
    else:
        author, rest = "", s
    author, rest = author.strip(), rest.strip()
    genre, title = "", rest
    if rest.endswith(")"):
        depth = 0
        for i in range(len(rest) - 1, -1, -1):  # pichon theke matching '(' khuji
            if rest[i] == ")":
                depth += 1
            elif rest[i] == "(":
                depth -= 1
                if depth == 0:
                    genre, title = rest[i + 1:-1].strip(), rest[:i].strip()
                    break
    return author, title, genre


def clean_body(seg):
    b = BODY.search(seg)
    body = b.group(1) if b else seg           # <body> na thakle (fragment) puro segment
    body = BLOCKS.sub("", body)
    body = WRAP_TAGS.sub("", body)            # fragment e thaka head/meta/title ityadi
    body = INLINE_STYLE.sub("", body)         # dark mode e inline color/background vange
    body = body.replace("\\'", "'").strip()   # source e thaka literal \' thik kori
    return balance_divs(body)


# ----------------------------------------------------------------- parsing
def parse_original(src):
    ms = list(SEP.finditer(src))
    cards = []
    for i, m in enumerate(ms):
        end = ms[i + 1].start() if i + 1 < len(ms) else len(src)
        author, title, genre = split_title(m.group(2))
        cards.append(dict(index=m.group(1), author=author, title=title,
                          genre=genre, body=clean_body(src[m.end():end])))
    return cards


CARD_RE = re.compile(r'<div\s+class="card[^"]*"\s+data-id="[^"]*"((?:\s+data-[a-z]+="[^"]*")*)\s*>', re.I)


def div_inner(text, start_pat, from_pos=0):
    m = re.compile(start_pat, re.S | re.I).search(text, from_pos)
    if not m:
        return None, from_pos
    depth, pos = 1, m.end()
    for t in DIV_TOK.finditer(text, pos):
        depth += -1 if t.group(1) else 1
        if depth == 0:
            return text[pos:t.start()], t.end()
    return text[pos:], len(text)


def parse_reformatted(src):
    cards, pos = [], 0
    while True:
        m = CARD_RE.search(src, pos)
        if not m:
            break
        attrs = {k: html.unescape(v) for k, v in re.findall(r'data-([a-z]+)="([^"]*)"', m.group(0))}
        inner, end = div_inner(src, re.escape(m.group(0)), m.start())
        body, _ = div_inner(inner or "", r'<div\s+class="details"\s*>')
        cards.append(dict(index=attrs.get("id", str(len(cards) + 1)), author=attrs.get("author", ""),
                          title=attrs.get("title", ""), genre=attrs.get("genre", ""),
                          body=(body or "").strip()))
        pos = end
    return cards

# ----------------------------------------------------------------- driver
def convert(src, path):
    if MARKER in src:
        cards = parse_reformatted(src)
        tm = TITLE.search(src)
        title = html.unescape(tm.group(1)).strip() if tm else path.stem
    else:
        cards = parse_original(src)
        title = path.stem.replace("_", " ")
    if not cards:
        return None, 0
    prefix = "idx_" + hashlib.md5(path.name.encode("utf-8")).hexdigest()[:8] + "_"
    return build_page(cards, title, prefix), len(cards)


def process_file(path, dry, backup):
    try:
        src = path.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return "SKIP (utf-8 noy)", 0
    page, n = convert(src, path)
    if page is None:
        return "SKIP (Index format na)", 0
    if page == src:
        return "UNCHANGED", n
    if dry:
        return "WOULD UPDATE", n
    if backup:
        shutil.copy2(path, path.with_suffix(path.suffix + ".bak"))
    tmp = path.with_name(path.name + ".tmp~")   # atomic in-place write, kono extra file thake na
    try:
        tmp.write_text(page, encoding="utf-8")
        tmp.replace(path)
    finally:
        if tmp.exists():
            tmp.unlink()
    return "UPDATED", n


def main():
    ap = argparse.ArgumentParser(description="'Index: N — Author >>> Title' type HTML - folder-wide in-place reformat")
    ap.add_argument("target", nargs="?", default=".", help="folder (default: current) ba single .html file")
    ap.add_argument("-r", "--recursive", action="store_true", help="subfolder gulo-o scan koro")
    ap.add_argument("--dry-run", action="store_true", help="kichu likhbe na, shudhu ki hobe dekhabe")
    ap.add_argument("--backup", action="store_true", help="update er age .bak rakho (default: kono backup/notun file na)")
    args = ap.parse_args()

    target = Path(args.target)
    if target.is_file():
        files = [target]
    elif target.is_dir():
        pat = "**/*" if args.recursive else "*"
        files = sorted(p for p in target.glob(pat) if p.is_file() and p.suffix.lower() in (".html", ".htm"))
    else:
        sys.exit(f"Path paoa jayni: {target}")
    if not files:
        sys.exit("Kono .html file paoa jayni.")

    counts = {}
    for f in files:
        status, n = process_file(f, args.dry_run, args.backup)
        key = status.split()[0]
        counts[key] = counts.get(key, 0) + 1
        print(f"[{status}] {f}" + (f" ({n} cards)" if n else ""))
    print("\nSummary:", ", ".join(f"{k}={v}" for k, v in counts.items()))


if __name__ == "__main__":
    main()
