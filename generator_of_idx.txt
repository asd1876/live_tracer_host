"""Generate a searchable, collapsible index.html for a folder tree.

Examples:
    python generate_index.py                      # index current folder
    python generate_index.py -r docs -o docs/index.html -t "Docs"
    python generate_index.py --open-depth 1 --ignore "*.tmp" --ignore drafts
    python generate_index.py --show-hidden --no-sizes

Extra ignore patterns can be put in a `.indexignore` file in the root
(one glob per line, '#' for comments). Patterns match a file/folder name
or its relative path, e.g.  *.log   build   docs/private
"""

]

def human_size(n):
    for unit in ("B", "KB", "MB", "GB"):
        if n < 1024 or unit == "GB":
            return f"{n:.0f} {unit}" if unit == "B" else f"{n:.1f} {unit}"
        n /= 1024


def load_ignore(root, extra, use_default):
    patterns = list(DEFAULT_IGNORE) if use_default else []
    patterns += extra
    ignore_file = root / ".indexignore"
    if ignore_file.is_file():
        for line in ignore_file.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if line and not line.startswith("#"):
                patterns.append(line.rstrip("/"))
    return patterns


def is_ignored(name, rel, patterns):
    return any(fnmatch.fnmatch(name, p) or fnmatch.fnmatch(rel, p) for p in patterns)


def scan(path, root, patterns, args, skip):
    """Return {'dirs': [...], 'files': [...]} for a directory, or None if empty."""
    try:
        entries = list(path.iterdir())
    except (PermissionError, OSError):
        return None
    dirs, files = [], []
    for item in entries:
        rel = item.relative_to(root).as_posix()
        if item.resolve() in skip:
            continue
        if is_ignored(item.name, rel, patterns):
            continue
        if item.name.startswith(".") and not args.show_hidden:
            continue
        try:
            if item.is_dir():
                if item.is_symlink() and not args.follow_symlinks:
                    continue
                sub = scan(item, root, patterns, args, skip)
                if sub is None or (not sub["dirs"] and not sub["files"] and not args.keep_empty):
                    continue
                sub.update(name=item.name, path=item)
                sub["count"] = len(sub["files"]) + sum(d["count"] for d in sub["dirs"])
                sub["size"] = sum(f["size"] for f in sub["files"]) + sum(d["size"] for d in sub["dirs"])
                sub["mtime"] = max([f["mtime"] for f in sub["files"]] + [d["mtime"] for d in sub["dirs"]] + [0])
                dirs.append(sub)
            elif item.is_file():
                st = item.stat()
                files.append({"name": item.name, "path": item, "size": st.st_size, "mtime": st.st_mtime})
        except OSError:
            continue
    key = lambda x: x["name"].lower()
    dirs.sort(key=key)
    files.sort(key=key)
    return {"dirs": dirs, "files": files}


def render(node, out_dir, args, depth=0):
    e = html.escape
    out = ["<ul>"]
    for d in node["dirs"]:
        rel = e(os.path.relpath(d["path"], out_dir).replace(os.sep, "/"), quote=True)
        is_open = " open" if depth < args.open_depth else ""
        meta = f'<span class="badge">{d["count"]}</span>'
        if args.sizes:
            meta += f'<span class="size">{human_size(d["size"])}</span>'
        out.append(
            f'<li class="dir" data-name="{e(d["name"].lower(), quote=True)}" '
            f'data-size="{d["size"]}" data-mtime="{d["mtime"]:.0f}" data-ext="">'
            f'<details{is_open}><summary><span class="ico">📁</span>'
            f'<span class="name">{e(d["name"])}</span>{meta}</summary>'
        )
        out.append(render(d, out_dir, args, depth + 1))
        out.append("</details></li>")
    for f in node["files"]:
        rel = os.path.relpath(f["path"], out_dir).replace(os.sep, "/")
        href = quote(rel)
        ext = f["name"].rsplit(".", 1)[-1].lower() if "." in f["name"] else ""
        icon = ICONS.get(ext, "📄")
        date = datetime.fromtimestamp(f["mtime"]).strftime("%Y-%m-%d")
        meta = ""
        if args.sizes:
            meta += f'<span class="size">{human_size(f["size"])}</span>'
        if args.dates:
            meta += f'<span class="date">{date}</span>'
        dl = (
            f'<a class="download" href="{href}" download="{e(f["name"], quote=True)}" '
            f'title="Download {e(f["name"], quote=True)}" '
            f'aria-label="Download {e(f["name"], quote=True)}">'
            f'<svg viewBox="0 0 16 16" width="14" height="14" aria-hidden="true">'
            f'<path fill="currentColor" d="M7.25 1.5h1.5v6.19l2.22-2.22 1.06 1.06L8 10.31 '
            f'3.97 6.53l1.06-1.06 2.22 2.22zM2 12h12v1.5H2z"/></svg>'
            f'<span>Download</span></a>'
            if args.downloads else ""
        )
        out.append(
            f'<li class="file" data-name="{e(f["name"].lower(), quote=True)}" '
            f'data-path="{e(rel.lower(), quote=True)}" data-size="{f["size"]}" '
            f'data-mtime="{f["mtime"]:.0f}" data-ext="{e(ext, quote=True)}">'
            f'<a class="link" href="{href}"><span class="ico">{icon}</span>'
            f'<span class="name">{e(f["name"])}</span></a>{meta}{dl}</li>'
        )
    out.append("</ul>")
    return "\n".join(out)

def generate(args):
    root = Path(args.root).resolve()
    output = Path(args.output).resolve() if args.output else root / "index.html"
    patterns = load_ignore(root, args.ignore, not args.no_default_ignore)
    tree = scan(root, root, patterns, args, skip={output})
    if tree is None:
        raise SystemExit(f"Cannot read directory: {root}")
    n_files = len(tree["files"]) + sum(d["count"] for d in tree["dirs"])
    n_dirs = 0

    def count_dirs(node):
        return sum(1 + count_dirs(d) for d in node["dirs"])

    n_dirs = count_dirs(tree)
    title = args.title or f"{root.name} index"
    summary = (f"{n_files} files in {n_dirs} folders · generated "
               f"{datetime.now():%Y-%m-%d %H:%M}")
    page = (TEMPLATE
            .replace("__TITLE__", html.escape(title))
            .replace("__SUMMARY__", html.escape(summary))
            .replace("__TREE__", render(tree, output.parent, args)))
    output.write_text(page, encoding="utf-8")
    print(f"Wrote {output} ({n_files} files, {n_dirs} folders)")


def main():
    p = argparse.ArgumentParser(description="Generate a searchable HTML index of a folder.")
    p.add_argument("-r", "--root", default=".", help="folder to index (default: .)")
    p.add_argument("-o", "--output", help="output file (default: <root>/index.html)")
    p.add_argument("-t", "--title", help="page title")
    p.add_argument("-i", "--ignore", action="append", default=[],
                   help="extra glob to ignore (repeatable)")
    p.add_argument("--no-default-ignore", action="store_true",
                   help="don't use the built-in ignore list")
    p.add_argument("--show-hidden", action="store_true", help="include dotfiles")
    p.add_argument("--keep-empty", action="store_true", help="show empty folders")
    p.add_argument("--follow-symlinks", action="store_true", help="descend into symlinked folders")
    p.add_argument("--open-depth", type=int, default=0,
                   help="folder levels expanded on load (default: 0 = all collapsed)")
    p.add_argument("--no-sizes", dest="sizes", action="store_false", help="hide file sizes")
    p.add_argument("--no-dates", dest="dates", action="store_false", help="hide modified dates")
    p.add_argument("--no-downloads", dest="downloads", action="store_false",
                   help="hide download buttons")
    generate(p.parse_args())


if __name__ == "__main__":
    main()
