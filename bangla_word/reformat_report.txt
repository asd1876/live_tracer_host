CARD_START = re.compile(r'<div\s+class="report-card"\s*>', re.I)
H2 = re.compile(r'<h2\s+class="heading"\s*>(.*?)</h2>', re.S | re.I)
STYLE = re.compile(r'<style\b.*?</style>', re.S | re.I)
TITLE = re.compile(r'<title>(.*?)</title>', re.S | re.I)
MAIN_TITLE = re.compile(r'<h1\s+class="main-title"\s*>(.*?)</h1>', re.S | re.I)


def div_block(text, start_pat, from_pos=0):
    """start_pat er prothom match theke tar matching </div> porjonto balanced block ferot dey.
    Return (inner_html, end_index) ba (None, from_pos)."""
    m = re.compile(start_pat, re.S | re.I).search(text, from_pos)
    if not m:
        return None, from_pos
    pos = m.end()
    depth = 1
    tok = re.compile(r'<(/?)div\b[^>]*>', re.I)
    for t in tok.finditer(text, pos):
        depth += -1 if t.group(1) else 1
        if depth == 0:
            return text[pos:t.start()], t.end()
    return text[pos:], len(text)


def split_cards(src):
    """Prottek report-card er balanced block alada kore."""
    cards = []
    pos = 0
    while True:
        m = CARD_START.search(src, pos)
        if not m:
            break
        inner, end = div_block(src, r'<div\s+class="report-card"\s*>', m.start())
        cards.append(inner)
        pos = end
    return cards


def parse_card(inner, idx):
    h = H2.search(inner)
    heading = h.group(1).strip() if h else f"{idx}. বিশ্লেষণ"
    # heading theke "N." number bad dei - amra nijera number dibo
    heading_text = re.sub(r'^\s*\d+\s*[.)]\s*', '', heading)

    orig, _ = div_block(inner, r'<div\s+class="original-box"\s*>')
    body, _ = div_block(inner, r'<div\s+class="analysis-content"\s*>')
    orig = (orig or "").strip()
    body = (body or "").strip()
    # global style leak kore (body, ul, li, strong...) -> card er vitore <style> felei dei
    body = STYLE.sub("", body)
    orig = STYLE.sub("", orig)
    return heading_text, orig, body

MARKER = 'name="generator" content="reformat_report"'
RF_CARD = r'<div\s+class="card[^"]*"\s+data-id="\d+"\s*>'
FOOTER = re.compile(r'\s*<div\s+class="footer-mark"\s*>.*?</div>\s*$', re.S | re.I)


def parse_reformatted(inner, idx):
    """Ager reformat kora card theke abar data ber kore (template refresh er jonno)."""
    h = H2.search(inner) or re.search(r'<h2\b[^>]*>(.*?)</h2>', inner, re.S)
    heading = h.group(1).strip() if h else f"{idx}. বিশ্লেষণ"
    orig, _ = div_block(inner, r'<div\s+class="orig"\s*>')
    body, _ = div_block(inner, r'<div\s+class="details"\s*>')
    body = FOOTER.sub("", body or "").strip()
    return heading, (orig or "").strip(), body


def split_generic(src, start_pat):
    cards, pos = [], 0
    while True:
        m = re.search(start_pat, src[pos:], re.S | re.I)
        if not m:
            break
        inner, end = div_block(src, start_pat, pos + m.start())
        cards.append(inner)
        pos = end
    return cards


def convert(src, name):
    """Return new html ba None (je file ta ei format er na)."""
    if MARKER in src:
        raw = split_generic(src, RF_CARD)
        cards = [parse_reformatted(c, i) for i, c in enumerate(raw, 1)]
    else:
        raw = split_cards(src)
        cards = [parse_card(c, i) for i, c in enumerate(raw, 1)]
    if not cards:
        return None, 0
    tm = TITLE.search(src)
    title = html.unescape(tm.group(1).strip()) if tm else Path(name).stem
    mm = MAIN_TITLE.search(src)
    main_title = mm.group(1).strip() if mm else html.escape(title)
    prefix = "rpt_" + hashlib.md5(name.encode("utf-8")).hexdigest()[:8] + "_"
    page = build_page(html.escape(re.sub(r'<.*?>', '', title)), main_title, cards, prefix)
    return page, len(cards)


def process_file(path, dry, backup):
    try:
        src = path.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return "SKIP (utf-8 noy)", 0
    page, n = convert(src, path.name)
    if page is None:
        return "SKIP (report format na)", 0
    if page == src:
        return "UNCHANGED", n
    if dry:
        return "WOULD UPDATE", n
    if backup:
        shutil.copy2(path, path.with_suffix(path.suffix + ".bak"))
    # atomic in-place write: temp file likhe shathe shathe replace kore dei (kono extra file thake na)
    tmp = path.with_name(path.name + ".tmp~")
    try:
        tmp.write_text(page, encoding="utf-8")
        tmp.replace(path)
    finally:
        if tmp.exists():
            tmp.unlink()
    return "UPDATED", n


def main():
    ap = argparse.ArgumentParser(description="Gemini analysis report HTML - folder-wide in-place auto-reformat")
    ap.add_argument("target", nargs="?", default=".", help="folder (default: current) ba single .html file")
    ap.add_argument("-r", "--recursive", action="store_true", help="subfolder gulo-o scan koro")
    ap.add_argument("--dry-run", action="store_true", help="kichu likhbe na, shudhu ki hobe dekhabe")
    ap.add_argument("--backup", action="store_true", help="update er age .bak rakho (default: kono backup/notun file na)")
    args = ap.parse_args()

    target = Path(args.target)
    if target.is_file():
        files = [target]
    elif target.is_dir():
        pat = "**/*" if args.recursive else "*"
        files = sorted(p for p in target.glob(pat) if p.is_file() and p.suffix.lower() in (".html", ".htm"))
    else:
        sys.exit(f"Path paoa jayni: {target}")
    if not files:
        sys.exit("Kono .html file paoa jayni.")

    counts = {}
    for f in files:
        status, n = process_file(f, args.dry_run, args.backup)
        key = status.split()[0]
        counts[key] = counts.get(key, 0) + 1
        extra = f" ({n} cards)" if n else ""
        print(f"[{status}] {f}{extra}")
    print("\nSummary:", ", ".join(f"{k}={v}" for k, v in counts.items()))


if __name__ == "__main__":
    main()
